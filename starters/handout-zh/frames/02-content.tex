\section{概述}

\begin{frame}{讲义}
  与幻灯同一套 \texttt{frames/}，用 Beamer \texttt{handout} + 四合一打印。
  \begin{itemize}
    \item 引擎：XeLaTeX + \texttt{ctexbeamer}
    \item 纸张：A4，每页 4 帧
  \end{itemize}
\end{frame}
