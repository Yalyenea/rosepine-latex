\newcommand{\makeposterbanner}{%
  \sbox{\posterbannerbox}{%
    \begin{minipage}[t]{\textwidth}
      \noindent{\color{coveraccent}\rule{\textwidth}{4pt}\par}
      \vspace{10pt}
      {\Huge\PosterHeadingWeight\color{covertxt}\PosterTitle\par}
      \vspace{5pt}
      \noindent
      \begin{minipage}[t]{0.70\textwidth}
        {\Large\color{coverpine}\PosterAuthors\par}
      \end{minipage}\hfill
      \begin{minipage}[t]{0.27\textwidth}
        \raggedleft{\small\color{rpsubtle}\PosterAffil\par}
      \end{minipage}\par
      \ifx\PosterTag\empty\else
        \vspace{10pt}
        \begin{tcolorbox}[posterstatement]
          \PosterTag
        \end{tcolorbox}
      \fi
      \vspace{10pt}
      \noindent{\color{rpoverlay}\rule{\textwidth}{2pt}\par}
      \vspace*{10pt}
    \end{minipage}%
  }%
  \noindent\usebox{\posterbannerbox}\par
  \setlength{\posterbodyheight}{%
    \dimexpr\textheight-\ht\posterbannerbox-\dp\posterbannerbox%
      -\parskip-\lineskip-4pt\relax}%
}
