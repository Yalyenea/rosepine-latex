\setbeamertemplate{footline}{%
  \leavevmode
  \hbox{%
    \begin{beamercolorbox}[wd=.72\paperwidth,ht=2.4ex,dp=1.1ex,leftskip=1.2em]{date}%
      \usebeamerfont{date}\insertshorttitle
      \ifx\insertsectionhead\empty\else
        \textcolor{rpmuted}{\quad/\quad}\insertsectionhead
      \fi
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=.28\paperwidth,ht=2.4ex,dp=1.1ex,rightskip=1.2em,right]{date}%
      \insertframenumber\,/\,\inserttotalframenumber
    \end{beamercolorbox}}%
  \vskip0pt%
}
